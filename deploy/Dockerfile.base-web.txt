# base-web prod build (multi-stage)
#
# context = ./base-web (由 docker-compose.prod.yml 指定)
# 完全不依賴 base-web/ 內 .dockerignore — 用 explicit COPY 列舉需要的檔/目錄,
# 自然排除 node_modules / dist / .git / .github / .vscode / README / CHANGELOG /
# lint config / .editorconfig / .gitattributes / .gitignore 等。
#
# 設計理由:user 要求「不改 base-web/ 內任何檔」,故 .dockerignore 不能放在 base-web/。

# === Stage 1: builder ===
FROM node:26-alpine AS builder
# 跳 corepack（用 npm install -g pnpm@10、不走 node 內建 corepack）——避 corepack wrap pnpm 觸發
# ERR_VM_DYNAMIC_IMPORT_CALLBACK_MISSING（前代 node 20.19 內建 corepack 踩過、詳 000-base-web-docker-bootstrap.md §4 第 3 輪 debug）。
# builder 與 dev **統一 node:26-alpine**（user 決定 2026-06-15、消除 20.19/26 分歧）;corepack 既被跳過、node 版本無關、prod build 已驗。
RUN npm install -g pnpm@10
WORKDIR /app

# 依賴清單先 copy → 利用 docker layer cache(改 source 不會 retrigger pnpm install)
COPY package.json pnpm-lock.yaml pnpm-workspace.yaml .npmrc ./
COPY packages ./packages
RUN pnpm install --frozen-lockfile

# source 後 copy
COPY build ./build
COPY public ./public
COPY src ./src
COPY .env .env.prod .env.test ./
COPY index.html vite.config.ts tsconfig.json uno.config.ts ./

# build-arg 接 host envvar(default 對齊現 base-web/.env.prod ApiFox Mock URL)
ARG VITE_SERVICE_BASE_URL=https://mock.apifox.cn/m1/3109515-0-default

# 寫 .env.prod.local — vite mode=prod precedence 最高、override .env.prod 同 key
# 注意:不用 ENV VITE_SERVICE_BASE_URL=$ARG,因 vite loadEnv 只讀 .env.* 不讀 process.env
RUN echo "VITE_SERVICE_BASE_URL=$VITE_SERVICE_BASE_URL" > .env.prod.local

# build:vite build --mode prod → dist/
RUN pnpm build

# 順便生成 nginx config 給 stage 2 COPY 用(避開 BuildKit heredoc 變數展開 quirk)
RUN mkdir -p /tmp/nginx && cat > /tmp/nginx/default.conf << 'NGINX_EOF'
server {
    listen 31079;
    root /usr/share/nginx/html;
    index index.html;
    location / {
        try_files $uri $uri/ /index.html;
    }
}
NGINX_EOF

# === Stage 2: runtime (nginx) ===
# 裸 nginx:alpine → pin 1.31.1-alpine（image pin 一致性 ＋ CVE-2026-42945、與 front-nginx 同版）
FROM nginx:1.31.1-alpine AS runtime
COPY --from=builder /app/dist /usr/share/nginx/html
COPY --from=builder /tmp/nginx/default.conf /etc/nginx/conf.d/default.conf
EXPOSE 31079
# 健檢打 / grep doctype(與 dev compose 層健檢同形、已實機驗證)——刻意偏離前代的
# /health.html 形:該檔在前代是 commit 進 base-web 分支的,rev3 本刀不動 base-web
# worktree(對齊檔頭設計理由＋constitution check「未觸 base-web inline」判定);
# 靜態 serve 下 / 回 index.html 與 health.html 等價佐證 nginx+檔案就緒
HEALTHCHECK --interval=10s --timeout=3s --start-period=5s --retries=3 \
    CMD wget -qO- http://127.0.0.1:31079/ | grep -qi "doctype" || exit 1
